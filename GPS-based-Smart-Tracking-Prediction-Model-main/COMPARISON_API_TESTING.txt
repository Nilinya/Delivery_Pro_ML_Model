PYTHON CLIENT EXAMPLE
=============================================================================

import requests
import json

# Server URL
BASE_URL = "http://localhost:5000"

# Base delivery data
base_delivery = {
    "Delivery_person_Age": 34,
    "Delivery_person_Ratings": 4.5,
    "Restaurant_latitude": 12.913041,
    "Restaurant_longitude": 77.683237,
    "Delivery_location_latitude": 13.043041,
    "Delivery_location_longitude": 77.813237,
    "Order_Date": "25-03-2022",
    "Time_Orderd": "19:45:00",
    "Time_Order_picked": "19:50:00",
    "Weatherconditions": "conditions Stormy",
    "Road_traffic_density": "Jam",
    "Vehicle_condition": 2,
    "Type_of_order": "Snack",
    "Type_of_vehicle": "scooter",
    "multiple_deliveries": 1,
    "Festival": "No",
    "City": "Metropolitian"
}

# Test all comparison types
comparison_types = [
    "distance_ranges",
    "vehicle_types",
    "weather_impact",
    "traffic_impact",
    "personnel_ratings"
]

for comp_type in comparison_types:
    payload = {
        "base_delivery": base_delivery,
        "comparison_type": comp_type
    }
    
    response = requests.post(
        f"{BASE_URL}/api/predict-comparison",
        json=payload,
        headers={"Content-Type": "application/json"}
    )
    
    if response.status_code == 200:
        result = response.json()
        print(f"\n{'='*60}")
        print(f"Comparison Type: {comp_type}")
        print(f"{'='*60}")
        print(f"Base Prediction: {result['base_prediction']} minutes")
        print(f"Fastest: {result['insights']['fastest_time']} min ({result['insights']['fastest_option']})")
        print(f"Slowest: {result['insights']['slowest_time']} min ({result['insights']['slowest_option']})")
        print(f"Time Variance: {result['insights']['time_variance']} min")
        print(f"Recommendation: {result['insights']['recommendation']}")
        
        print(f"\nAll Scenarios:")
        for i, comp in enumerate(result['comparisons'], 1):
            print(f"  {i}. {comp['label']:<50} → {comp['predicted_time']:>6.1f} min")
    else:
        print(f"Error for {comp_type}: {response.status_code}")
        print(response.json())

TROUBLESHOOTING
=============================================================================

Problem: "Model not loaded"
Solution: Run `python train_model.py` first to train and save the model

Problem: KeyError on specific field
Solution: Ensure all required fields are present in base_delivery (check COMPARISON_API_DOCS.txt)

Problem: "Invalid comparison_type"
Solution: Use only: distance_ranges, vehicle_types, weather_impact, traffic_impact, personnel_ratings

Problem: Slow response (>2 sec)
Solution: This is OK for distance_ranges (slowest), faster for others

Problem: NaN or null in results
Solution: Check that base_delivery coordinates are valid (not 0.0, 0.0)
